Run it from your code

Get an API token in the Apify Console (Settings → Integrations), then:

curl -X POST \
  "https://api.apify.com/v2/acts/plainrows~keyword-search-volume/run-sync-get-dataset-items?token=$APIFY_TOKEN" \
  -H 'Content-Type: application/json' \
  -d '{
  "keywords": [
    "coffee maker",
    "espresso machine",
    "french press",
    "cold brew coffee",
    "coffee grinder",
    "pour over coffee",
    "moka pot",
    "milk frother",
    "decaf coffee",
    "coffee beans"
  ],
  "mode": "volume",
  "country": "US"
}'
from apify_client import ApifyClient

client = ApifyClient("<YOUR_APIFY_TOKEN>")
run = client.actor("plainrows/keyword-search-volume").call(run_input={'keywords': ['coffee maker',
              'espresso machine',
              'french press',
              'cold brew coffee',
              'coffee grinder',
              'pour over coffee',
              'moka pot',
              'milk frother',
              'decaf coffee',
              'coffee beans'],
 'mode': 'volume',
 'country': 'US'})
for row in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(row)
